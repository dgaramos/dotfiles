# zoxide
z <dir>           # zoxide: jump to frecent directory (fuzzy match)
z -               # zoxide: jump to previous directory
zi                # zoxide: interactive jump with fzf
zoxide query -l   # zoxide: list all tracked directories

# fzf
Ctrl+T            # fzf: fuzzy insert file path at cursor
Ctrl+R            # fzf: fuzzy search command history
Alt+C             # fzf: fuzzy cd into a directory

# direnv
direnv allow      # direnv: allow .envrc in current directory
direnv deny       # direnv: revoke .envrc permissions
direnv reload     # direnv: reload current .envrc

# eza

# bat
bat -l <lang>     # bat: force syntax highlight for a specific language
bat --list-themes # bat: list all available themes
bat --plain       # bat: output without decorations

# ripgrep (rg / rgrep)
rg <pattern>      # rg: recursive grep with color
rg -l <pattern>   # rg: list only files that match
rg -t py <pat>    # rg: grep only files of a given type
rg --hidden <pat> # rg: include hidden files in search

# fd / ff
ff <name>         # fd: find files by name
ff -e py          # fd: find files by extension
ff -t d <name>    # fd: find directories only
ff --hidden <n>   # fd: include hidden files

# gh (GitHub CLI)
gh repo view      # gh: view current repo in browser
gh run list       # gh: list recent CI runs
gh run watch      # gh: watch a running CI job
gh auth status    # gh: show current auth status

# sshm (SSH manager)
sshm list                # sshm: list configured SSH hosts
sshm add                 # sshm: interactive wizard to add a host
sshm edit                # sshm: open ~/.ssh/config in $EDITOR
sshm copy-id <host>      # sshm: copy public key to server
sshm keygen              # sshm: wizard to generate a new key pair

# local-env (machine-local env vars)
local-env set N V        # local-env: set a machine-local env variable
local-env unset N        # local-env: remove a machine-local env variable
local-env get N          # local-env: print current value of a variable
local-env list           # local-env: list all managed variables
local-env edit           # local-env: open state file in $EDITOR

# localz (machine-local shell config)
localz edit              # localz: open local.zsh in $EDITOR
localz show              # localz: print contents of local.zsh
localz list              # localz: list aliases and functions in local.zsh
localz add N CMD         # localz: append an alias to local.zsh
