# zoxide
z <dir>           # zoxide: jump to frecent directory (fuzzy match)
z -               # zoxide: jump to previous directory

zoxide query -l   # zoxide: list all tracked directories

# fzf
Ctrl+T            # fzf: fuzzy insert file path at cursor
Ctrl+R            # fzf: fuzzy search command history
Alt+C             # fzf: fuzzy cd into a directory

# direnv
direnv allow      # direnv: allow .envrc in current directory
direnv deny       # direnv: revoke .envrc permissions
direnv reload     # direnv: reload current .envrc

# bat

bat --list-themes # bat: list all available themes

# ripgrep (rg / rgrep)
rg <pattern>      # rg: recursive grep with color
rg -l <pattern>   # rg: list only files that match
rg -t py <pat>    # rg: grep only files of a given type
rg --hidden <pat> # rg: include hidden files in search

# fd / ff
ff <name>         # fd: find files by name
ff -e py          # fd: find files by extension
ff -t d <name>    # fd: find directories only
ff --hidden <n>   # fd: include hidden files
